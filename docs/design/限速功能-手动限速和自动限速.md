# 限速功能-手动限速和自动限速

 


# 手动限速

配置接口：&quot;/config/request_limit&quot;  

```plaintext
curl -XPOST -H "content-type: application/json" -d'
{
  "request_limit_enabled": true,
  "read_request_limit_count": 1000,
  "write_request_limit_count": 500
}
' http://${VEARCH_URL}/config/request_limit

配置参数信息存储在etcd中key->/router/request_limit_config value->配置参数值
配置参数修改会同步修改etcd中保存的元数据信息，router节点监听到元数据信息更新时同步更新
```


| 字段标识 | 类型 | 描述 |
|---|---|---|
| request_limit_enabled | `bool` | 是否开启限速功能。 |
| read_request_limit_count | `float64` | 集群qps限制 |
| write_request_limit_count | `float64` | 集群tps限制 |

```plaintext
func HttpLimitMiddleware(docService docService) gin.HandlerFunc {
    return func(c *gin.Context) {
        if c.Request.ContentLength > 0 && gctuner.CheckMemoryLimitExceed(uint64(c.Request.ContentLength)) {
            msg := fmt.Sprintf("memory usage have reached limit %d before execute request", entity.ConfigInfo.MemoryLimitConfig.RouterMemoryLimit)
            log.Error(msg)

            c.Abort()
            return
        }

        Request := strings.TrimPrefix(c.Request.RequestURI, "/document/")

        switch Request {
        case "upsert", "delete":
            if !entity.WriteLimiter.Allow() {

                response.New(c).JsonError(errors.NewErrInternal(fmt.Errorf(msg)))
                c.Abort()
                return
            }
        case "query", "search":
            if !entity.ReadLimiter.Allow() {
                msg := fmt.Sprintf("document read request too frequency, have reached limit %d", entity.ReadLimiter.Burst())
                log.Error(msg)
                response.New(c).JsonError(errors.NewErrInternal(fmt.Errorf(msg)))

                return
            }
        }

        c.Next()
    }
}
```

# 自动限速

master根据ps节点的cpu使用率自动对集群的tps/qps进行限速。

配置接口：

```plaintext
/config/cpu_limit

curl -XPOST -H "content-type: application/json" -d'
{
  "cpu_high_watermark": 90,
  "cpu_low_watermark": 50,
  "monitor_interval": 60,
  "limit_status": 1,
}
' http://${VEARCH_URL}/config/cpu_limit

```


| 字段名称 | 类型 | 描述 |
|---|---|---|
| `cpu_high_watermark` | `float64` | 存在ps节点的 CPU 使用率超过这个值时，触发集群限流，默认值95。 |
| `cpu_low_watermark` | `float64` | 所有ps节点的 CPU 使用率都低于这个值时，逐步放开集群限流，默认值70。 |
| `monitor_interval` | `int` | 基于CPU探测是否限流 的时间间隔（以秒为单位），默认值60。 |
| `limit_servers` | `map[string]int` | 内部字段，记录了所有被限制的节点和它们的限制状态，当节点的cpu使用率低于`cpu_low_watermark时`将被从限制列表中移除。 |
| `limit_status` | `int` | 是否开启自动限速。1 表示开启，2 表示关闭。 |
| `is_limited` | `bool` | 内部字段，当前集群是否被限速。 |
| `reduce_percent` | `int` | 内部字段，触发限流时，tps/qps下降的百分比。 |
| `last_updated` | `time.Time` | 内部字段，最近一次更新限流状态的时间戳。 |

```plaintext
    data, err := os.ReadFile("/sys/fs/cgroup/cpu/cpuacct.usage")
    if err != nil {
        log.Error("read cpu usage failed, err:[%s]", err.Error())
        return 0, err
```

## master-自动限速

```plaintext
1.master服务启动时，开启CPULimitService协程，自动探测
func (s *CPULimitService) Start() {
    if s.running {
        log.Error("CPU limit service is already running")
        return
    }

    s.running = true
    go s.monitorLoop()
}

每隔10s执行一次检测
func (s *CPULimitService) monitorLoop() {
    defer func() {
        if err := recover(); err != nil {
            log.Error("CPU limit service monitorLoop panic: %v", err)
            go s.monitorLoop()
        }
    }()

    s.monitorInterval = entity.CPULimitInfo.MonitorInterval
    s.ticker = time.NewTicker(10 * time.Second)
    defer s.ticker.Stop()

    for {
        select {
        case <-s.ctx.Done():
            log.Info("CPU limit service stopped")
            return
        case <-s.ticker.C:
            s.checkPSNodesCPUUsage()
        }
    }
}

真正的探测执行函数
func (s *CPULimitService) checkPSNodesCPUUsage() {
    // only main master do check work
        // 只有master的主节点会执行探测
    if !s.isMaster() {
        return
    }
    mc := s.client.Master()

        //配置参数的探测时间间隔修改时，修改CPULimitService的monitorInterval
    if s.monitorInterval != entity.CPULimitInfo.MonitorInterval {
        s.monitorInterval = entity.CPULimitInfo.MonitorInterval
    }
        //请求关闭cpu自动限速后，修改CPULimitService的相应状态，并直接返回
    if entity.CPULimitInfo.LimitStatus == entity.LimitDisabled {
        if entity.CPULimitInfo.IsLimited {
            log.Debug("Removing rate limit from all routers")
            ctx, cancel := context.WithTimeout(s.ctx, 10*time.Second)
            defer cancel()

            if _, err := s.removeRateLimit(ctx, 0); err == nil {
                entity.CPULimitInfo.IsLimited = false
                entity.CPULimitInfo.LimitedServers = make(map[entity.NodeID]int)
                entity.CPULimitInfo.ReducePercent = entity.DefaultReducePercent
                entity.CPULimitInfo.LastUpdated = time.Now()
            } else {
                log.Error("Failed to remove rate limit from all Router nodes: %v", err)
            }
        }
        return
    }

    //enhance flag
    enhanceLimit := false
    changed := false
    curPsMaxCpuUsage := 0.0

    //一个cpu自动限速探测周期内，会执行多次对ps节点cpu使用率的检查
    
    if s.monitorCycles < s.monitorInterval {
        s.monitorCycles = s.monitorCycles + 10
        items := s.client.Master().Cache().PSCpuCache().Items()

        //每隔10s检测一次各个ps节点的cpu使用率，获取所有ps节点的cpu使用率的最大值-curPsMaxCpuUsage
        //主要检查是否有ps节点的cpu使用率是否在[CPULowWatermark,CPUHighWatermark]区间内，只要在探测周期内
        //至少存在一次某个ps节点的cpu使用率在这个区间，将该节点放在CPULimitInfo.LimitedServers中
        //某个ps节点的cpu使用率大于CPUHighWatermark时，设置enhanceLimit为true
        //（这一步主要是为了防止cpu使用率频繁波动，只要在探测周期内存在有节点的cpu使用率在[CPULowWatermark,CPUHighWatermark]区间，
        // 且enhanceLimit为false，则保持当前tps/qps限速设置）
        for nodeIdStr, item := range items {
            if cpuUsage, ok := item.Object.(float64); ok {

                nodeId := entity.NodeID(cast.ToUint64(nodeIdStr))

                if cpuUsage > curPsMaxCpuUsage {
                    curPsMaxCpuUsage = cpuUsage
                }

                if cpuUsage > entity.CPULimitInfo.CPULowWatermark {
                    if _, exists := entity.CPULimitInfo.LimitedServers[nodeId]; !exists {
                        entity.CPULimitInfo.LimitedServers[nodeId] = 1
                        changed = true
                    }
                    if cpuUsage > entity.CPULimitInfo.CPUHighWatermark {
                        //log.Debug("PS node %d CPU usage is high: %.2f%% more than %.2f%%, marking for rate limit", nodeId, cpuUsage, entity.CPULimitInfo.CPUHighWatermark)
                        enhanceLimit = true
                    }
                }
            }
        }
    }

    //执行完上述检查后，还不到参数设置的自动探测时间间隔（CPULimitInfo.MonitorInterval），则直接返回，否则继续执行
    if s.monitorInterval > s.monitorCycles {
        return
    }

    ctx, cancel := context.WithTimeout(s.ctx, 10*time.Second)
    defer cancel()

    //当有节点满足cpu在[CPULowWatermark,CPUHighWatermark]区间内时，判断enhanceLimit是否为true
    //若为false,则保持当前限速设置，若为true，则进行进一步限速
    if len(entity.CPULimitInfo.LimitedServers) > 0 {
        if enhanceLimit {
            log.Debug("At least one PS node have high CPU usage more than %.2f%%, applying enhanced rate limit to all routers", entity.CPULimitInfo.CPUHighWatermark)
            //目前限速幅度较小，设置为下降1%，cpu被打满时，较小的限速幅度效果就很明显
            if entity.CPULimitInfo.ReducePercent > 50 {
                entity.CPULimitInfo.ReducePercent = entity.CPULimitInfo.ReducePercent - 1
                changed = true
            } else {
                entity.CPULimitInfo.ReducePercent = 50
            }
            ////根据当前ReducePercent进一步限制qps/tps
            s.applyRateLimit(ctx)
        }

        entity.CPULimitInfo.IsLimited = true
    } else if len(entity.CPULimitInfo.LimitedServers) == 0 && entity.CPULimitInfo.IsLimited {
        percent := 0.0
        //当所有ps节点的cpu使用率都小于CPULowWatermark时，根据curPsMaxCpuUsage的范围设置限速放开时的涨幅
        if curPsMaxCpuUsage+20 < entity.CPULimitInfo.CPULowWatermark {
            percent = 10
        } else if curPsMaxCpuUsage+10 < entity.CPULimitInfo.CPULowWatermark {
            percent = 5
        } else {
            percent = 1
        }
        log.Debug("All PS nodes have normal CPU usage less than %.2f%%, Directly, removing rate limit from all routers", curPsMaxCpuUsage)
        entity.CPULimitInfo.ReducePercent = entity.DefaultReducePercent
        if isLimited, err := s.removeRateLimit(ctx, percent); err == nil {
            if isLimited != entity.CPULimitInfo.IsLimited {
                entity.CPULimitInfo.IsLimited = isLimited
                changed = true
            }
        } else {
            log.Error("Failed to remove rate limit from all Router nodes: %v", err)
        }
    }

        //entity.CPULimitInfo有更新时，更新元数据信息
    if changed {
        entity.CPULimitInfo.LastUpdated = time.Now()

        jsonData, err := json.Marshal(entity.CPULimitInfo)
        if err != nil {
            log.Error("Failed to marshal CPU limit state: %v", err)
            return
        }

        if err := mc.Update(s.ctx, entity.CPULimitConfigKey(entity.CpuLimitConfigKey), jsonData); err != nil {
            log.Error("Failed to update CPU limit state: %v", err)
            return
        }
    }

    s.monitorCycles = 0
    clear(entity.CPULimitInfo.LimitedServers)
}


//根据各个router节点上报的qps/tps，进行限速设置
func (s *CPULimitService) applyRateLimit(ctx context.Context) {
    totalQueryRate, totalWriteRate, _ := s.GetRouterQpsTotalStats()

    reducePercent := float64(entity.CPULimitInfo.ReducePercent) / 100.0

    cfg := &entity.RequestLimitCfg{
        RequestLimitEnabled: true,
        TotalReadLimit:      totalQueryRate * reducePercent,
        TotalWriteLimit:     totalWriteRate * reducePercent,
    }
    log.Debug("Router limit config: %+v", cfg)

    err := s.configService.ModifyRequestLimitCfg(ctx, cfg)
    if err != nil {
        log.Error("Failed to apply rate limit to all Router nodes: %v", err)
        return
    }
}

//逐步放开限速，直到达到DefaultReadRequestLimitCount时不再调整
func (s *CPULimitService) removeRateLimit(ctx context.Context, percent float64) (bool, error) {
    totalQueryRate, totalWriteRate, _ := s.GetRouterQpsTotalStats()

    isLimited := false
    reducePercent := percent / 100.0
    if totalQueryRate == 0 || reducePercent == 0 {
        totalQueryRate = entity.DefaultReadRequestLimitCount
    } else {
        totalQueryRate = totalQueryRate * (1 + reducePercent)
        if totalQueryRate > entity.DefaultReadRequestLimitCount {
            totalQueryRate = entity.DefaultReadRequestLimitCount
        } else {
            isLimited = true
        }
    }

    if totalWriteRate == 0 || reducePercent == 0 {
        totalWriteRate = entity.DefaultWriteRequestLimitCount
    } else {
        totalWriteRate = totalWriteRate * (1 + reducePercent)
        if totalWriteRate > entity.DefaultWriteRequestLimitCount {
            totalWriteRate = entity.DefaultWriteRequestLimitCount
        } else {
            isLimited = true
        }
    }

    cfg := &entity.RequestLimitCfg{
        RequestLimitEnabled: true,
        TotalReadLimit:      totalQueryRate,
        TotalWriteLimit:     totalWriteRate,
    }

    err := s.configService.ModifyRequestLimitCfg(ctx, cfg)
    if err != nil {
        log.Error("Failed to remove rate limit from all Router nodes: %v", err)
        return isLimited, err
    }

    log.Debug("Removed rate limit %+v from all Router nodes", cfg)
    return isLimited, nil
}
```
